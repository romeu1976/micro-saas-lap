  <script>
    (function () {
      const form = document.getElementById("calc-form");
      const clearBtn = document.getElementById("clear-btn");
      const panel = document.getElementById("result-panel");
      const highlight = document.getElementById("result-highlight");
      const details = document.getElementById("result-details");
      const copyBtn = document.getElementById("copy-result-btn");

      if (!form || !panel || !highlight || !details) return;

      form.addEventListener("submit", function (event) {
        event.preventDefault();
        // Substituir pela lógica específica da calculadora
        highlight.textContent = "—";
        details.textContent = "Preencha os dados válidos para visualizar o cálculo.";
        panel.hidden = false;
      });

      if (clearBtn) {
        clearBtn.addEventListener("click", function () {
          form.reset();
          panel.hidden = true;
          highlight.textContent = "";
          details.textContent = "";
        });
      }

      if (copyBtn && navigator.clipboard) {
        copyBtn.addEventListener("click", async function () {
          try {
            const text = `${highlight.textContent} — ${details.textContent}`;
            await navigator.clipboard.writeText(text.trim());
            copyBtn.textContent = "Copiado!";
            setTimeout(() => {
              copyBtn.textContent = "Copiar resultado";
            }, 1800);
          } catch (_) {
            // Ignora falha silenciosa se permissão for negada
          }
        });
      }
    })();
  </script>
